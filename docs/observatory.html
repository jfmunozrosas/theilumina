Official inequality indicators from Eurostat, the World Bank and the Spanish Statistical Office (INE), and links to other inequality observatories

Modified

November 16, 2026

This observatory gathers public inequality indicators for European and world countries and for the Spanish regions. Eurostat and World Bank data are downloaded live from the official sources every time the page is opened, so they always show the latest release; INE data are downloaded every time the website is updated. Each chart can be explored by hovering over the points, and the data can be downloaded as CSV.

Why Theil indices are rarely published, and what this observatory offers

Official statistics usually publish the Gini index and the S80/S20 ratio, but seldom Theil indices, even though Theil indices can be decomposed into inequality between and within groups. This observatory shows:

  • Theil T_0 (mean log deviation), published by the World Bank’s Poverty and Inequality Platform (PIP) for about 170 countries.
  • An approximation of T_0 and T_1 for EU countries, computed from the income shares of the ten deciles published by Eurostat (EU-SILC). Because it only uses inequality between deciles, it is a lower bound of the true index; inequality within each decile is ignored.

All these figures are point estimates based on samples. As shown by this project, Theil estimators can be biased and their margin of error can be large when incomes are highly skewed. The theil R package will provide bias-corrected estimates and confidence intervals from microdata.

function jsonstat(js) {
  // Flattens a JSON-stat 2.0 dataset (Eurostat API) into an array of records
  const ids = js.id, size = js.size, dims = js.dimension;
  const codes = ids.map(d => {
    const idx = dims[d].category.index;
    const arr = Array.isArray(idx) ? idx : Object.keys(idx).sort((a, b) => idx[a] - idx[b]);
    return arr;
  });
  const labels = ids.map(d => dims[d].category.label || {});
  const out = [];
  const vals = js.value;
  const keys = Array.isArray(vals) ? vals.map((v, i) => i).filter(i => vals[i] != null) : Object.keys(vals).map(Number);
  for (const i of keys) {
    let rem = i; const rec = {};
    for (let d = ids.length - 1; d >= 0; d--) {
      const c = codes[d][rem % size[d]];
      rec[ids[d]] = c;
      if (ids[d] === "geo") rec.country = labels[d][c] || c;
      rem = Math.floor(rem / size[d]);
    }
    rec.value = +vals[i];
    out.push(rec);
  }
  return out;
}
pref = ["TOTAL", "T", "SHARE", "EUR", "PC", "A"]
function keepPreferred(recs, keepDims) {
  // For dimensions other than geo/time (and keepDims) keep a single category
  if (!recs.length) return recs;
  const dimsAll = Object.keys(recs[0]).filter(k => !["geo", "time", "value", "country", ...keepDims].includes(k));
  let r = recs;
  for (const d of dimsAll) {
    const cats = [...new Set(r.map(x => x[d]))];
    const chosen = pref.find(p => cats.includes(p)) ?? cats[0];
    r = r.filter(x => x[d] === chosen);
  }
  return r;
}
async function eurostat(code, query = "") {
  const url = `https://ec.europa.eu/eurostat/api/dissemination/statistics/1.0/data/${code}?format=JSON&lang=EN${query}`;
  const res = await fetch(url);
  if (!res.ok) {
    // Eurostat returns the reason of the error in the body (JSON-stat "error" field)
    let msg = `HTTP ${res.status}`;
    try {
      const js = await res.json();
      const err = Array.isArray(js.error) ? js.error[0] : js.error;
      if (err && err.label) msg += ` – ${err.label}`;
    } catch (e) {}
    throw new Error(`Eurostat ${code}: ${msg}`);
  }
  return jsonstat(await res.json());
}

async function decileShares() {
  // Income shares by decile (ilc_di01). Shares do not depend on the currency, so only EUR is requested;
  // this keeps the request small. If the joint request is rejected, deciles are requested one by one.
  const base = "&indic_il=SHARE&currency=EUR&sinceTimePeriod=2005";
  try {
    return await eurostat("ilc_di01", base + deciles.map(q => `&quantile=${q}`).join(""));
  } catch (e1) {
    const parts = await Promise.all(deciles.map(q => eurostat("ilc_di01", `${base}&quantile=${q}`)));
    return parts.flat();
  }
}

// ---------- Download Eurostat data ----------
euroData = {
  // Each dataset is loaded independently: if one of them fails, the others are still shown
  const [gini, s80, dec] = await Promise.allSettled([eurostat("ilc_di12"), eurostat("ilc_di11"), decileShares()]);
  const rows = [], errors = [];
  if (gini.status === "fulfilled") {
    for (const r of keepPreferred(gini.value, [])) rows.push({geo: r.geo, country: r.country, year: +r.time, indicator: "Gini index", value: r.value});
  } else errors.push(gini.reason.message);
  if (s80.status === "fulfilled") {
    for (const r of keepPreferred(s80.value, [])) rows.push({geo: r.geo, country: r.country, year: +r.time, indicator: "S80/S20 ratio", value: r.value});
  } else errors.push(s80.reason.message);
  if (dec.status === "fulfilled") {
    // Theil approximations from decile shares (grouped data, lower bound)
    const d = keepPreferred(dec.value, ["quantile"]);
    const groups = d3.group(d, r => r.geo, r => r.time);
    for (const [geo, byYear] of groups) for (const [time, recs] of byYear) {
      if (recs.length !== 10) continue;
      const tot = d3.sum(recs, r => r.value);
      const s = recs.map(r => r.value / tot).filter(v => v > 0);
      if (s.length !== 10) continue;
      const T1 = d3.sum(s, v => v * Math.log(10 * v));
      const T0 = d3.sum(s, v => 0.1 * Math.log(0.1 / v));
      rows.push({geo, country: recs[0].country, year: +time, indicator: "Theil T1 (decile approx.)", value: T1});
      rows.push({geo, country: recs[0].country, year: +time, indicator: "Theil T0 (decile approx.)", value: T0});
    }
  } else errors.push(dec.reason.message);
  const out = rows.filter(r => EU.has(r.geo) && Number.isFinite(r.value));
  return {ok: out.length > 0, rows: out, errors, error: errors.join("; "),
          indicators: [...new Set(out.map(r => r.indicator))]};
}

Theil approximations computed by THEILUMINA from the decile shares s_k: T_1 \approx \sum_k s_k \ln(10\,s_k) and T_0 \approx \sum_k 0.1 \ln(0.1/s_k).

```{ojs}
function jsonstat(js) {
  // Flattens a JSON-stat 2.0 dataset (Eurostat API) into an array of records
  const ids = js.id, size = js.size, dims = js.dimension;
  const codes = ids.map(d => {
    const idx = dims[d].category.index;
    const arr = Array.isArray(idx) ? idx : Object.keys(idx).sort((a, b) => idx[a] - idx[b]);
    return arr;
  });
  const labels = ids.map(d => dims[d].category.label || {});
  const out = [];
  const vals = js.value;
  const keys = Array.isArray(vals) ? vals.map((v, i) => i).filter(i => vals[i] != null) : Object.keys(vals).map(Number);
  for (const i of keys) {
    let rem = i; const rec = {};
    for (let d = ids.length - 1; d >= 0; d--) {
      const c = codes[d][rem % size[d]];
      rec[ids[d]] = c;
      if (ids[d] === "geo") rec.country = labels[d][c] || c;
      rem = Math.floor(rem / size[d]);
    }
    rec.value = +vals[i];
    out.push(rec);
  }
  return out;
}
```

```{ojs}
  const rows = [], errors = [];
  if (gini.status === "fulfilled") {
    for (const r of keepPreferred(gini.value, [])) rows.push({geo: r.geo, country: r.country, year: +r.time, indicator: "Gini index", value: r.value});
  } else errors.push(gini.reason.message);
  if (s80.status === "fulfilled") {
    for (const r of keepPreferred(s80.value, [])) rows.push({geo: r.geo, country: r.country, year: +r.time, indicator: "S80/S20 ratio", value: r.value});
  } else errors.push(s80.reason.message);
  if (dec.status === "fulfilled") {
    // Theil approximations from decile shares (grouped data, lower bound)
    const d = keepPreferred(dec.value, ["quantile"]);
    const groups = d3.group(d, r => r.geo, r => r.time);
    for (const [geo, byYear] of groups) for (const [time, recs] of byYear) {
      if (recs.length !== 10) continue;
      const tot = d3.sum(recs, r => r.value);
      const s = recs.map(r => r.value / tot).filter(v => v > 0);
      if (s.length !== 10) continue;
      const T1 = d3.sum(s, v => v * Math.log(10 * v));
      const T0 = d3.sum(s, v => 0.1 * Math.log(0.1 / v));
      rows.push({geo, country: recs[0].country, year: +time, indicator: "Theil T1 (decile approx.)", value: T1});
      rows.push({geo, country: recs[0].country, year: +time, indicator: "Theil T0 (decile approx.)", value: T0});
    }
  } else errors.push(dec.reason.message);
  const out = rows.filter(r => EU.has(r.geo) && Number.isFinite(r.value));
  return {ok: out.length > 0, rows: out, errors, error: errors.join("; "),
          indicators: [...new Set(out.map(r => r.indicator))]};
}
```

## Spain: inequality by region (INE)

The Spanish Statistical Office (INE) publishes the **Gini index** and the **S80/S20 ratio** of equivalised disposable income for Spain and its 19 regions (autonomous communities and cities), from the Living Conditions Survey (ECV). The data are downloaded from the INE API every time the website is rendered.

```{r}
#| label: ine-download
#| include: false
# INE table 67246: Desigualdad en la distribución de ingresos (S80/S20 y coeficiente de Gini) por CCAA (ECV)
ine_file <- "data/ine_67246.csv"
dir.create("data", showWarnings = FALSE)
ine <- tryCatch({
  tmp <- tempfile(fileext = ".csv")
  download.file("https://www.ine.es/wstempus/csv_sc/es/DATOS_TABLA/67246", tmp, quiet = TRUE, mode = "wb")
  x <- read.csv2(tmp, fileEncoding = "UTF-8-BOM", check.names = FALSE, colClasses = "character")
  names(x) <- c("region", "indicador", "year", "value")
  x$value  <- as.numeric(gsub(",", ".", gsub(".", "", x$value, fixed = TRUE), fixed = TRUE))
  x$year   <- as.integer(x$year)
  x$region <- sub("^[0-9]+ ", "", x$region)
  x$region[x$region == "Total Nacional"] <- "Spain (total)"
  lab <- c("Gini" = "Gini index",
           "Desigualdad (S80/S20)" = "S80/S20 ratio",
           "Gini (con alquiler imputado)" = "Gini index (with imputed rent)",
           "Distribución de la renta S80/S20 (alquiler imputado)" = "S80/S20 ratio (with imputed rent)")
  x$indicator <- unname(lab[x$indicador])
  x <- x[!is.na(x$value) & !is.na(x$indicator), c("region", "indicator", "year", "value")]
  write.csv(x, ine_file, row.names = FALSE, fileEncoding = "UTF-8")
  x

  # If INE cannot be reached, use the last downloaded copy (if any)
  if (file.exists(ine_file)) read.csv(ine_file, fileEncoding = "UTF-8") else
```
