## Where we work

The team works at the Faculty of Economics and Business of the University of Granada and at partner universities in Spain, the United Kingdom, Italy and Portugal. Click on a marker to see the institution and its researchers.

```{=html}
<div id="team-map" class="map-box" role="region" aria-label="Map of the institutions of the team"></div>
<script>
(function () {
  const sites = [
    {name: "Faculty of Economics and Business, University of Granada", city: "Granada, Spain", lat: 37.1938, lon: -3.5985, main: true,
     who: "Principal investigators and most of the team (Quantitative Methods, Statistics and Operations Research, Applied Economics, International and Spanish Economics)"},
    {name: "University of Edinburgh Business School", city: "Edinburgh, United Kingdom", lat: 55.9441, lon: -3.1868, who: "Ferran Vendrell-Herrero"},
    {name: "University of Glasgow", city: "Glasgow, United Kingdom", lat: 55.8721, lon: -4.2882, who: "Cecilia Tortajada Quiroz"},
    {name: "Nova School of Business and Economics, NOVA University Lisbon", city: "Carcavelos (Lisbon), Portugal", lat: 38.6781, lon: -9.3263, who: "Emanuel Gomes"},
    {name: "University of Modena and Reggio Emilia", city: "Modena, Italy", lat: 44.6380, lon: 10.9230, who: "Tindara Addabbo"},
    {name: "University of León", city: "León, Spain", lat: 42.6140, lon: -5.5590, who: "Mª Nazaret Ibáñez Rueda"},
    {name: "University of Valladolid", city: "Valladolid, Spain", lat: 41.6620, lon: -4.7150, who: "Samara López Ruiz"}
  ];
  function draw() {
    if (typeof L === "undefined") { setTimeout(draw, 200); return; }
    const map = L.map("team-map", {scrollWheelZoom: false});
    L.tileLayer("https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png", {
      maxZoom: 18, attribution: '&copy; <a href="https://www.openstreetmap.org/copyright">OpenStreetMap</a> contributors'
    }).addTo(map);
    const pts = [];
    for (const s of sites) {
      const m = L.circleMarker([s.lat, s.lon], {radius: s.main ? 11 : 8, color: "#ffffff", weight: 2,
        fillColor: s.main ? "#E07B20" : "#1F4E7A", fillOpacity: 0.95}).addTo(map);
      m.bindPopup(`<b>${s.name}</b><br>${s.city}<br><span style="color:#555">${s.who}</span>`);
      pts.push([s.lat, s.lon]);
    }
    map.fitBounds(pts, {padding: [30, 30]});
  }
  draw();
})();
</script>
```
